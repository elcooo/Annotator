git commands:
    # 1. Start from the latest main branch
    git checkout main
    git pull origin main

    # 2. Create and switch to a new branch for your feature
    git checkout -b feature/my-test

    # 3. Work on your changes (edit files, run tests, etc.)
    #    When you want to see what’s changed:
    git status
    git diff

    all good? : yes -> go to step 4
        # 4. If the branch looks good, commit and push (same as before)
        git add .
        git commit -m "Describe the successful changes"
        git push -u origin feature/my-test

        (Optional) Merge to main if you want:
        git checkout main
        git pull origin main
        git merge feature/my-test
        git push origin main

    all good? : no -> go to step 5

        # 5. If the branch does NOT look good and you want to abandon it:
        git checkout main                     # switch back to main
        git branch -D feature/my-test      # delete the local branch
        git push origin --delete feature/my-test  # delete it on GitHub (if pushed)
